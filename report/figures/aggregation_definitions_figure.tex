  \begin{scope}[shift={(#1,#2)}]

\newcommand{\cgrid}[4]{%
  \begin{scope}[shift={(#2,#3)}]
    \foreach \c/\r/\v in {#4}{
      \fill[#1!\v!white] (\c*\px,\r*\px) rectangle ++(\cz,\cz);
      \draw[cell]        (\c*\px,\r*\px) rectangle ++(\cz,\cz);
    }
  \end{scope}}

% rounded score bar: #1 accent, #2 x, #3 ycenter, #4 fill width (frame width 1.78)
\newcommand{\sbar}[4]{%
