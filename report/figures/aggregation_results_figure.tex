% Aggregation results figure: two-stage story.
%   Left  panel : experiment 010 head-only sweep (mean vs max vs softmax-mean),
%                 matched grid_4x4 ts448 ov0.5 -- the clean, same-checkpoint
%                 comparison that justifies dropping plain mean.
%   Right panel : final i002 four-block checkpoint (max vs softmax-mean, plus
%                 softmax-mean + logit adjustment). Plain mean was NOT submitted
%                 here, so it is intentionally absent from the right panel.
%
% Numbers: docs/submissions_archive/submission_manifest.csv (public + private
% Macro F1). Each bar is the best-public / best-private selection for that mode.
%

%   \usepackage{pgfplots}\pgfplotsset{compat=1.18}
%   \usepgfplotslibrary{groupplots}
%
% NB: ybar / bar width are set per-axis (inside each \nextgroupplot), not in the
% shared groupplot options, because this pgfplots version does not recognise
% bar width in the top-level groupplot key list.
%
% Include in a figure* float (see sections/experiments.tex).
