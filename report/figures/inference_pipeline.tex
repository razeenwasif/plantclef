% ── Column of stages (top to bottom) ─────────────────────────────────────
\node[io]    (img)    {Quadrat image \\
                       \small\itshape (raw RGB)};
\node[stage] (tiles)  [below=of img]    {Multi-scale grid tiling \\
                                          {\scriptsize $4{\times}4$ grid · scales $\{1.0, 0.8\}$}};
\node[stage] (veg)    [below=of tiles]  {ExG vegetation filter \\
                                          {\scriptsize drop tiles with $<\!15\%$ green pixels}};
\node[emph]  (fwd)    [below=of veg]    {BioCLIP 2.5 \,ViT-H/14 (last $4$ blocks unfrozen) \\
                                          {\scriptsize forward at $224$\,px \quad+\quad forward at $336$\,px}};
\node[stage] (agg)    [below=of fwd]    {Per-tile Bayesian aggregation \\
                                          {\scriptsize $w_t = \exp(-H_t)\cdot \mathrm{ExG}_t$, softmax-weighted mean}};
\node[stage] (xres)   [below=of agg]    {Cross-resolution fusion \\
                                          {\scriptsize weighted mean of $224$\,px / $336$\,px probabilities}};
\node[stage] (la)     [below=of xres]   {Class-prior logit adjustment \\
                                          {\scriptsize $z' = z - \tau \log \tilde{\pi}_s$, $\tau{=}0.25$}};
\node[stage] (thr)    [below=of la]     {Sigmoid \,+\, adaptive threshold \\
                                          {\scriptsize emit $\{s : \sigma(z'_s) > T\}$, $T{=}0.03$, $k_{\min}{=}2$}};
\node[io]    (out)    [below=of thr]    {Species set \\
                                          \small\itshape (Macro-F1 0.418)};

% ── Sequential arrows ────────────────────────────────────────────────────
\draw[flow] (img)   -- (tiles);
\draw[flow] (tiles) -- (veg);
\draw[flow] (veg)   -- (fwd);
\draw[flow] (fwd)   -- (agg);
\draw[flow] (agg)   -- (xres);
\draw[flow] (xres)  -- (la);
\draw[flow] (la)    -- (thr);
\draw[flow] (thr)   -- (out);

% ── Scope groupings on the left ─────────────────────────────────────────
\begin{pgfonlayer}{background}

  \node[scope, fit=(fwd) (agg) (xres)] (s-model) {};
  \node[scopelabel, anchor=east] at ($(s-model.north west)+(-2pt,-7pt)$) {Visual posterior};

  \node[scope, fit=(la) (thr)] (s-post) {};
  \node[scopelabel, anchor=east] at ($(s-post.north west)+(-2pt,-7pt)$) {Post-processing};
\end{pgfonlayer}

% ── Side-band: parameter callouts on the right ──────────────────────────

  {Laplace-smoothed prior};
