%% Compatibility shim: colortbl >= v1.0k references \insert@pcolumn for
%% paragraph (p / m / b / X) columns, but array.sty < 2024 only defines
%% \insert@column. On installations with a mixed package set (e.g. MiKTeX
%% with colortbl 2024/10/29 v1.0k and array 2023/10/16 v2.5g) p-columns and
%% tabularx X-columns abort with "Undefined control sequence". Alias the
%% missing token to the equivalent older one.
\makeatletter
\@ifundefined{insert@pcolumn}{\let\insert@pcolumn\insert@column}{}
\makeatother
